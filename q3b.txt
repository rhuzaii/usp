Write a C program to implement ls -li command which lists the files in a specified directory. Your program should print at least 5 attributes.

#include <stdio.h>
#include <dirent.h>
#include <sys/stat.h>
#include <string.h>
#include <pwd.h>
#include <grp.h>
#include <time.h>

void print_permissions(mode_t mode) {
    printf((S_ISDIR(mode)) ? "d" : "-");

    printf((mode & S_IRUSR) ? "r" : "-");
    printf((mode & S_IWUSR) ? "w" : "-");
    printf((mode & S_IXUSR) ? "x" : "-");

    printf((mode & S_IRGRP) ? "r" : "-");
    printf((mode & S_IWGRP) ? "w" : "-");
    printf((mode & S_IXGRP) ? "x" : "-");

    printf((mode & S_IROTH) ? "r" : "-");
    printf((mode & S_IWOTH) ? "w" : "-");
    printf((mode & S_IXOTH) ? "x" : "-");
}

int main(int argc, char *argv[]) {
    DIR *dir;
    struct dirent *entry;
    struct stat st;
    char path[1024];

    if (argc < 2) {
        printf("Usage: %s <directory>\n", argv[0]);
        return 1;
    }

    dir = opendir(argv[1]);
    if (!dir) {
        perror("opendir failed");
        return 1;
    }

    while ((entry = readdir(dir)) != NULL) {

        // Build full path
        snprintf(path, sizeof(path), "%s/%s", argv[1], entry->d_name);

        if (stat(path, &st) == -1) {
            perror("stat failed");
            continue;
        }

        // Print inode
        printf("%llu ", (unsigned long long)st.st_ino);

        // Print permissions
        print_permissions(st.st_mode);

        // Links
        printf(" %hu ", (unsigned short)st.st_nlink);

        // Owner
        struct passwd *pw = getpwuid(st.st_uid);
        printf("%s ", pw->pw_name);

        // Group
        struct group *gr = getgrgid(st.st_gid);
        printf("%s ", gr->gr_name);

        // Size
        printf("%lld ", (long long)st.st_size);

        // Time
        printf("%.12s ", ctime(&st.st_mtime) + 4);

        // Filename
        printf("%s\n", entry->d_name);
    }

    closedir(dir);
    return 0;
}
