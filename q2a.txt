Write a C program to display various details of a file using stat structure (At least 5 fields).

#include <stdio.h>
#include <sys/stat.h>
#include <unistd.h>
#include <time.h>

int main(int argc, char *argv[]) {
    struct stat st;

    // Check if filename is provided
    if (argc < 2) {
        printf("Usage: %s <filename>\n", argv[0]);
        return 1;
    }

    // Call stat using user input
    if (stat(argv[1], &st) == -1) {
        perror("stat failed");
        return 1;
    }

    printf("File: %s\n", argv[1]);
    printf("Inode Number: %ld\n", st.st_ino);
    printf("File Size: %ld bytes\n", st.st_size);
    printf("Number of Links: %ld\n", st.st_nlink);
    printf("User ID: %d\n", st.st_uid);
    printf("Last Modified: %s", ctime(&st.st_mtime));

    return 0;
}

nano q2a.c
gcc q2a.c -o q2a
vi file.txt ( hello world )
