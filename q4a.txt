Write a C program to demonstrate the creation of soft links and hard links

#include <stdio.h>
#include <unistd.h>

int main() {

    // Create hard link
    if (link("file1.txt", "hardlink.txt") == -1) {
        perror("Hard link failed");
    } else {
        printf("Hard link created successfully\n");
    }

    // Create soft link
    if (symlink("file1.txt", "softlink.txt") == -1) {
        perror("Soft link failed");
    } else {
        printf("Soft link created successfully\n");
    }

    return 0;
}


nano q4a.c
gcc q4a.c -o q4a
vi file1.txt (hello world)
./q4a
