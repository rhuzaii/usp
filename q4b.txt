Write a C program to
i. To create a child process.
ii. The child should execute an interpreter file by passing a few
arguments
iii. Create an interpreter file that has the path of echoall.c file and
pass one argument
iv. Create echoall.c file which prints the arguments received from
both child process and interpreter file.

nano echoall.c

#include <stdio.h>

int main(int argc, char *argv[]) {
    printf("Arguments received:\n");

    for (int i = 0; i < argc; i++) {
        printf("argv[%d] = %s\n", i, argv[i]);
    }

    return 0;
}

gcc echoall.c -o echoall

nano interpreter.sh

#!/bin/sh

./echoall "$1" "$2"

chmod +x interpreter.sh

nano q4b.c
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid;

    pid = fork();

    if (pid < 0) {
        perror("fork failed");
        return 1;
    }

    if (pid == 0) {
        // Child process
        execl("./interpreter.sh", "interpreter.sh", "Hello", "World", (char *)NULL);

        // Only runs if exec fails
        perror("exec failed");
    } else {
        // Parent process
        wait(NULL);
        printf("Parent finished\n");
    }

    return 0;
}

gcc q4b.c -o q4b
