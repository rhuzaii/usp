Write a C program to demonstrate the usage of umask and chmod functions.”

#include <stdio.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>

int main() {
    int fd;

    // Step 1: Set umask
    umask(022);

    // Step 2: Create file
    fd = open("testfile.txt", O_CREAT | O_WRONLY, 0666);
    if (fd < 0) {
        perror("File creation failed");
        return 1;
    }

    close(fd);

    printf("File created with umask applied.\n");

    // Step 3: Change permission using chmod
    if (chmod("testfile.txt", 0777) == -1) {
        perror("chmod failed");
        return 1;
    }

    printf("Permissions changed to 0777 using chmod.\n");

    return 0;
}
