Write a C program to demonstrate the use of wait() and waitpid() system calls.

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <stdlib.h>

int main() {
    pid_t pid1, pid2;
    int status;

    // First child
    pid1 = fork();

    if (pid1 == 0) {
        printf("Child 1 (PID: %d) running\n", getpid());
        sleep(2);
        printf("Child 1 exiting\n");
        exit(1);
    }

    // Second child
    pid2 = fork();

    if (pid2 == 0) {
        printf("Child 2 (PID: %d) running\n", getpid());
        sleep(4);
        printf("Child 2 exiting\n");
        exit(2);
    }

    // Parent process

    // wait() → waits for any child
    pid_t finished = wait(&status);
    printf("wait(): Child %d finished\n", finished);

    // waitpid() → wait for specific child (pid2)
    waitpid(pid2, &status, 0);
    printf("waitpid(): Child %d finished\n", pid2);

    return 0;
}
